Complete Saved Groups Implementation

Implementation Overview

This page contains the complete working implementation of the Saved Groups functionality that needs to be integrated into the sidebar.js file.

The implementation includes:

  • Loading saved groups from Chrome storage
  • Saving current tab groups with custom names
  • Restoring saved groups in current or new window
  • Deleting saved groups
  • Import/Export functionality
  • Proper error handling and user feedback

// 1. Replace loadSavedGroups method:
async loadSavedGroups() {
  try {
    const result = await chrome.storage.local.get("savedGroups");
    this.savedGroups = result.savedGroups || [];
    console.log('Loaded saved groups:', this.savedGroups.length);
  } catch (error) {
    console.error("Error loading saved groups:", error);
    this.savedGroups = [];
  }
}

// 2. Replace saveCurrentGroupWithTags method:
async saveCurrentGroupWithTags(groupId) {
  try {
    const group = this.groups.find((g) => g.id === groupId);
    const tabs = this.tabs.filter((tab) => tab.groupId === groupId);

    if (!group || tabs.length === 0) {
      console.error('Group not found or empty');
      return;
    }

    const name = prompt("Enter a name for this saved group:", group.title || "Unnamed Group");
    if (!name) return;

    const savedGroup = {
      id: Date.now().toString(),
      name: name,
      color: group.color,
      tabs: tabs.map((tab) => ({
        title: tab.title,
        url: tab.url,
        favIconUrl: tab.favIconUrl,
        pinned: tab.pinned || false
      })),
      savedAt: new Date().toISOString(),
      tabCount: tabs.length
    };

    // Load current saved groups
    await this.loadSavedGroups();
    
    // Add new group
    this.savedGroups.unshift(savedGroup);
    
    // Save to storage
    await chrome.storage.local.set({ savedGroups: this.savedGroups });
    
    console.log(`Saved group "${name}" with ${tabs.length} tabs`);

    // Switch to saved groups tab to show the newly saved group
    this.switchTabSection("saved");
    
    // Render saved groups
    this.renderSavedGroups();
  } catch (error) {
    console.error("Error saving group:", error);
    alert("Error saving group: " + error.message);
  }
}

// 3. Replace renderSavedGroups method:
renderSavedGroups() {
  const container = document.getElementById("saved-groups-content");

  if (!this.savedGroups || this.savedGroups.length === 0) {
    container.innerHTML =
      '
No saved groups
';
    return;
  }

  container.innerHTML = this.savedGroups
    .map(
      (group) => `
    

      

        

          

          ${group.name}
          ${group.tabCount || group.tabs.length}
          
        

        

          
          
          
          
        

      

      

        ${group.tabs
          .slice(0, 5)
          .map(
            (tab) => `
          

            '
            }" alt="">
            

              
${tab.title}

              
${new URL(tab.url).hostname}

            

          

        `
          )
          .join("")}
        ${group.tabs.length > 5 ? `
... and ${group.tabs.length - 5} more tabs
` : ''}
      

    

  `
    )
    .join("");
}

// 4. Replace deleteSavedGroup method:
async deleteSavedGroup(savedGroupId) {
  if (confirm("Are you sure you want to delete this saved group?")) {
    try {
      this.savedGroups = this.savedGroups.filter((g) => g.id !== savedGroupId);
      await chrome.storage.local.set({ savedGroups: this.savedGroups });
      this.renderSavedGroups();
      console.log("Deleted saved group:", savedGroupId);
    } catch (error) {
      console.error("Error deleting saved group:", error);
      alert("Error deleting group: " + error.message);
    }
  }
}

// 5. Replace restoreSavedGroup method:
async restoreSavedGroup(savedGroupId, inNewWindow = false) {
  try {
    const savedGroup = this.savedGroups.find((g) => g.id === savedGroupId);
    if (!savedGroup) {
      console.error('Saved group not found');
      return;
    }

    let windowId;
    if (inNewWindow) {
      const newWindow = await chrome.windows.create();
      windowId = newWindow.id;
      // Remove the default tab
      const tabs = await chrome.tabs.query({ windowId });
      if (tabs.length > 0) {
        await chrome.tabs.remove(tabs[0].id);
      }
    } else {
      const currentWindow = await chrome.windows.getCurrent();
      windowId = currentWindow.id;
    }

    // Create tabs
    const tabIds = [];
    for (const tabData of savedGroup.tabs) {
      const tab = await chrome.tabs.create({
        url: tabData.url,
        windowId,
        active: false,
        pinned: tabData.pinned || false
      });
      tabIds.push(tab.id);
    }

    // Group the tabs
    if (tabIds.length > 0) {
      const groupId = await chrome.tabs.group({ tabIds });
      await chrome.tabGroups.update(groupId, {
        title: savedGroup.name,
        color: savedGroup.color || "blue",
      });
    }

    console.log(`Restored group "${savedGroup.name}" with ${tabIds.length} tabs`);
    this.debouncedRefresh();
  } catch (error) {
    console.error("Error restoring saved group:", error);
    alert("Error restoring group: " + error.message);
  }
}
